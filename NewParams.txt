# ============================================================
#  Конфіг реклами (Brainrots) — формат: ключ=значення
# ============================================================
#  Правила формату:
#   - Один параметр на рядок: КЛЮЧ=ЗНАЧЕННЯ
#   - Рядки, що починаються з '#' або '//' — коментарі (ігноруються)
#   - Порожні рядки ігноруються
#   - Регістр ключів НЕ має значення (minsecondsbetweenads = minSecondsBetweenAds)
#   - Значення enum можна писати назвою (interstitialMode=Mixed)
#     або номером (interstitialMode=2) — і те, і те зрозуміє парсер
#   - Підтримуються лише ключі з цього файлу. Невідомий або некоректний
#     ключ просто ігнорується — значення з інспектора залишається.
#
#  ВАЖЛИВО: з сервера міняється ЛИШЕ ПОВЕДІНКА (таймери, тексти, тип реклами,
#  кути банерів). Чутливі налаштування — ad unit ID, середовище (Test/Production),
#  tagForUnderAgeOfConsent, debugForceEEA, testDeviceId — з сервера НЕ читаються,
#  вони задаються ТІЛЬКИ в інспекторі. Це зроблено навмисно: помилка в файлі
#  не зможе зламати згоду (COPPA) або підмінити ваші рекламні ID.
#
#  Куди покласти цей файл: у грі адреса береться з поля NetParams.fileURL:
#    https://raw.githubusercontent.com/p1ggsy/tber-privacy/refs/heads/main/params.txt
#  Тобто: закомітьте цей файл у репозиторій p1ggsy/tber-privacy
#  (гілка main, шлях params.txt) — і гра завантажить його при старті.
#
#  Згенеровано 14.08.2026 з поточних значень інспектора сцени Playground.
# ============================================================


# ---------- СИСТЕМА РЕКЛАМИ ----------
# true  = СТАРА система: банер зверху, реклама лише за запитом ShowFullAd()/ShowRewardedAd()
# false = НОВА система (за замовчуванням): авто-таймер, банери по кутах, App Open,
#         Rewarded Interstitial з інтро-екраном
# Зміна застосовується до ініціалізації реклами; якщо реклама вже ініціалізована —
# спрацює при наступному запуску гри (у консолі буде попередження).
oldSystem=true


# ---------- АВТО-ТАЙМЕР повноекранної реклами (сек) ----------
# Через скільки секунд після попереднього показу таймер знову
# спробує показати рекламу. Випадкове число в діапазоні [min, max].
minSecondsBetweenAds=20
maxSecondsBetweenAds=25

# УНІВЕРСАЛЬНИЙ КУЛДАУН: жодна реклама (авто чи ручний виклик)
# не з'явиться частіше ніж раз на цю кількість секунд.
ADS_RATE=30


# ---------- РЕКЛАМА З НАГОРОДОЮ + інтро-екран ----------
# Час показу інтро-екрана з кнопкою "Пропустити" перед
# Rewarded Interstitial (обов'язково за політикою AdMob), сек.
introScreenSeconds=3
introRewardText=Watch a short ad to continue!
introSkipButtonText=Skip


# ---------- ТИП авто-реклами ----------
# Що показує автоматичний таймер:
#   RewardedInterstitial — з нагородою, з'являється інтро-екран "Пропустити"
#   PlainInterstitial    — просто міжсторінкова, показується одразу, без екрана
#   Mixed                — випадково між двома типами (шанс нагороди нижче)
interstitialMode=Mixed
mixedRewardedChancePercent=50


# ---------- РЕКЛАМА ЗАВАНТАЖЕННЯ (App Open) ----------
# Показувати при холодному старті гри / при поверненні з фонового режиму?
showAppOpenOnColdStart=true
showAppOpenOnResume=true

# Скільки секунд чекаємо завантаження App Open при старті
# (якщо не встиг — гра просто продовжується без реклами).
appOpenLoadTimeoutSeconds=5

# Мінімальний інтервал між показами App Open (хв).
appOpenMinIntervalMinutes=3

# Пауза між повторними спробами завантаження будь-якої реклами (сек).
retryDelaySeconds=10


# ---------- ОРІЄНТАЦІЯ ТА БАНЕРИ ----------
# Примусово перевертати гру в ландшафтну орієнтацію?
forceLandscapeOrientation=true

# До якого кута екрана кріпити лівий/правий банер:
#   TopLeft=0, TopRight=1, BottomLeft=2, BottomRight=3
# (уВАГА: зараз у сцені стоїть 0 і 1 — це ВЕРХНІ кути!)
leftBannerCorner=TopLeft
rightBannerCorner=TopRight


# ---------- ОНОВЛЕННЯ КОНФІГА ----------
# Якщо > 0 — конфіг перезавантажується кожні N хвилин (зміни на сервері
# підхопляться БЕЗ перезапуску гри). 0 — лише один раз при старті.
remoteConfigRefreshMinutes=0
